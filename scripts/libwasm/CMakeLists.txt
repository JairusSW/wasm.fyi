function(ladybird_lib name fs_name)
    add_library(${name} STATIC ${SOURCES} ${GENERATED_SOURCES})
    if(NOT name STREQUAL "AK")
        target_link_libraries(${name} PRIVATE AK)
    endif()
    if("EXPLICIT_SYMBOL_EXPORT" IN_LIST ARGN)
        string(TOUPPER "${fs_name}" macro)
        generate_export_header(${name} EXPORT_MACRO_NAME "${macro}_API" EXPORT_FILE_NAME Export.h)
    endif()
endfunction()
